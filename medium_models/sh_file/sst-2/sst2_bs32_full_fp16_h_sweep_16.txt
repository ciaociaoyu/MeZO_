#!/bin/bash
#SBATCH --job-name=sst2_bs32_full_fp16_h_sweep_16
#SBATCH --partition=gpu_p
#SBATCH --array=0-1%2
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=2
#SBATCH --mem=100G
#SBATCH --gres=gpu:H100:1
#SBATCH --time=24:00:00
#SBATCH --output=jobs/%x_%A_%a.out

set -eu
set -o pipefail 2>/dev/null || true

source ~/.bashrc
conda activate ciao
ml jq

cd /home/jy03364/MeZO/medium_models
mkdir -p jobs

export TASK=SST-2
export MODEL=roberta-large
export K=16
export SEEDS="16"
export DATA_SEED=16
export DATASET_MODE=full

TASK_WORKSPACE="/home/jy03364/MeZO/medium_models/sh_file/sst-2/full_fp16_h_sweep_16_workspace"
mkdir -p "${TASK_WORKSPACE}/result" "${TASK_WORKSPACE}/launch_meta"

export BS=32
export LR=1e-6
export WD=0
export STEP=50000
export EVAL_STEP=5000
export DATALOADER_SHUFFLE=False

export USE_H=False
export USE_C=False
export ZO_PROBE_EVERY=200
export ZO_PROBE_NUM_SEEDS=16
export ZO_PROBE_LOG_CSV=True
export PRECISIONS="fp16"

export SWEEP_NAME="SST-2-bs32-full-fp16-h-sweep-seed16"
export RESULT_ROOT="${TASK_WORKSPACE}/result"
export SUMMARY_FILE="${RESULT_ROOT}/${SWEEP_NAME}/summary.jsonl"
export FORCE_ALL_CASES=1

cat > "${TASK_WORKSPACE}/launch_meta/job_${SLURM_JOB_ID:-manual}_array_${SLURM_ARRAY_TASK_ID:-na}.txt" <<EOF
task=${TASK}
bs=${BS}
seed=${SEEDS}
data_seed=${DATA_SEED}
dataset_mode=${DATASET_MODE}
workspace=${TASK_WORKSPACE}
EOF

echo "SLURM_ARRAY_TASK_ID=${SLURM_ARRAY_TASK_ID}"
echo "Task workspace: ${TASK_WORKSPACE}"
echo "Summary file: ${SUMMARY_FILE}"

case "${SLURM_ARRAY_TASK_ID}" in
    0)
        export EPS_LIST_OVERRIDE="1e-9 3e-9 1e-8 3e-8 1e-7 3e-7 1e-6 3e-6"
        ;;
    1)
        export EPS_LIST_OVERRIDE="1e-5 3e-5 1e-4 3e-4 1e-3 3e-3 1e-2"
        ;;
    *)
        echo "Unexpected SLURM_ARRAY_TASK_ID=${SLURM_ARRAY_TASK_ID}"
        exit 1
        ;;
esac

echo "EPS chunk: ${EPS_LIST_OVERRIDE}"

bash scripts/exp_sst5_h_precision_sweep.sh
