#!/bin/bash
#SBATCH --job-name=singleh_sst5_truedir
#SBATCH --partition=gpu_p
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=2
#SBATCH --mem=100G
#SBATCH --gres=gpu:L4:1
#SBATCH --time=24:00:00


#SBATCH --output=jobs/%x_%A_%a.out
#SBATCH --mail-user=jichao.yu@uga.edu
#SBATCH --mail-type=ALL

ml jq

source ~/.bashrc
conda activate ciao

cd /home/jy03364/MeZO/medium_models

# =========================
# (A) 固定 h 单独跑：关闭 h-probe（避免额外计算）
# =========================
export H_PROBE=0
unset H_PROBE_EVERY
unset H_PROBE_NDIR
unset H_PROBE_NB
unset H_PROBE_UNIT_U
unset H_PROBE_SAVE_PARAMS
unset H_PROBE_LOG_EACH_H
unset H_PROBE_OUT
unset H_PROBE_HLIST
unset H_PROBE_MIN
unset H_PROBE_MAX

# =========================
# (B) h 列表：每个 array task 取一个 eps/h
# =========================
H_LIST=( \
  1e-3 \
)

EPS="${H_LIST[$SLURM_ARRAY_TASK_ID]}"
echo "SLURM_ARRAY_TASK_ID=${SLURM_ARRAY_TASK_ID}"
echo "Running TRUE directional derivative mode; EPS(h) tag = ${EPS}"

# =========================
# (C) 训练设置
# =========================
export STEP=50000
export EVAL_STEP=5000

# 让 tag 里也带上 eps，便于你聚合 log
export EXTRA_TAG="singleh-truedir-eps${EPS}"

# 强烈建议：给每个 eps 一个独立 job_name（否则输出目录会冲突/覆盖）
JOB_NAME="sst5_singleh_truedir_eps${EPS}"

# 结果根目录（建议和 finite-diff 结果分开，避免混淆）
RESULT_ROOT="/home/jy03364/MeZO/medium_models/sh_file/sst5_bs8/research_loss_h_zaoshewng/single_h_true_dir"

# =========================
# (D) 开跑：固定 h（zero_order_eps=EPS）
#     关键新增：--zo_use_true_directional_derivative
# =========================
TASK=sst-5 \
K=16 \
SEED=16 \
DATA_SEED=16 \
BS=8 \
LR=1e-6 \
EPS=1e-3 \
MODEL=roberta-large \
USE_H=False \
USE_C=False \
DATALOADER_SHUFFLE=True \
STEP=$STEP \
EVAL_STEP=$EVAL_STEP \
EXTRA_TAG=$EXTRA_TAG \
bash mezo.sh \
  --result_root "$RESULT_ROOT" \
  --job_name "$JOB_NAME" \
  --zo_use_true_directional_derivative
